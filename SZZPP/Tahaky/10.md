# Otázka 10: R – zpracování dat, kód

```r
library(tidyverse)
library(readxl)

# načtení dat
df <- read_csv("StudentsPerformance.csv")
# nebo Excel:
# df <- read_excel("data.xlsx", sheet = 1)

# nový sloupec – průměr
df <- df %>%
  mutate(avg_score = (math.score + reading.score + writing.score) / 3)

# filtrování dle textu (obsahuje řetězec / regex)
df_hs <- df %>%
  filter(str_detect(parental.level.of.education, "high school"))

# seskupení + průměr
summary_tbl <- df_hs %>%
  group_by(gender, race.ethnicity) %>%
  summarise(prumer = mean(avg_score), .groups = "drop")

# do širokého formátu (kategorie gender jako sloupce)
wide_tbl <- summary_tbl %>%
  pivot_wider(names_from = gender, values_from = prumer)

# zpět do dlouhého formátu
long_tbl <- wide_tbl %>%
  pivot_longer(cols = -race.ethnicity, names_to = "gender", values_to = "prumer")

# řazení a výběr sloupců
df_sorted <- df %>%
  arrange(desc(avg_score)) %>%
  select(gender, avg_score)

# spojování tabulek
# spojena <- left_join(tabulka1, tabulka2, by = "klic")

# --- grafy ---
# boxplot
ggplot(df_hs, aes(x = race.ethnicity, y = avg_score)) +
  geom_boxplot() +
  labs(title = "Rozložení avg_score dle skupiny")

# histogram
ggplot(df_hs, aes(x = avg_score)) +
  geom_histogram(binwidth = 5) +
  labs(title = "Histogram avg_score")

# sloupcový graf
ggplot(summary_tbl, aes(x = race.ethnicity, y = prumer, fill = gender)) +
  geom_col(position = "dodge")
```

## Rychlá reference – co kam patří
| Operace | Funkce |
|---|---|
| nový/upravený sloupec | `mutate()` |
| filtrování řádků | `filter()`, `str_detect()` pro text |
| seskupení + agregace | `group_by() %>% summarise()` |
| řazení | `arrange()` (`desc()` pro sestupně) |
| výběr sloupců | `select()` |
| široký ↔ dlouhý formát | `pivot_wider()` / `pivot_longer()` |
| spojení tabulek | `left_join()`, `inner_join()`, `full_join()` |
| kontingenční tabulka | `pivot_wider()` po `summarise()`, nebo `table()` |
